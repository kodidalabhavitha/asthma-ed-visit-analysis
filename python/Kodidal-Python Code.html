import pandas as pd
import matplotlib.pyplot as plt

# Load the dataset
data = pd.read_csv('/Users/bhavithakodidala/Desktop/untitled folder 2/asthma-ed-visit-rates_2019.csv')

# Filter data for the years 2015 to 2019
data_filtered = data[data['Year'].isin([2015, 2016, 2017, 2018, 2019])]

# Group data by year and calculate the average ED visit rate for each year
annual_data = data_filtered.groupby('Year')['Rate'].mean().reset_index()

# Plot the data
plt.figure(figsize=(10, 6))
plt.plot(annual_data['Year'], annual_data['Rate'], marker='o', linestyle='-', color='b')
plt.title('Asthma ED Visit Rates (2015-2019)')
plt.xlabel('Year')
plt.ylabel('Average ED Visit Rate')
plt.grid(True)
plt.xticks(annual_data['Year'])
plt.show()

# Perform a basic statistical analysis
rate_change = annual_data['Rate'].iloc[-1] - annual_data['Rate'].iloc[0]
if rate_change > 0:
    print("Asthma ED visit rates have risen from 2015 to 2019.")
else:
    print("Asthma ED visit rates have lowered from 2015 to 2019.")

import pandas as pd
import matplotlib.pyplot as plt

# Load the dataset
file_path = '/Users/bhavithakodidala/Desktop/untitled folder 2/asthma-ed-visit-rates_2019.csv'  # Replace with your dataset path
data = pd.read_csv(file_path)

# Filter relevant columns
columns_needed = ['Geography', 'Year', 'Rate']  # Update with actual column names
filtered_data = data[columns_needed]

# Group by geography and calculate the mean rate for each region
region_avg_rates = filtered_data.groupby('Geography')['Rate'].mean().reset_index()
region_avg_rates = region_avg_rates.sort_values(by='Rate', ascending=False)

# Calculate the statewide average
statewide_avg = filtered_data['Rate'].mean()

# Add a comparison column
region_avg_rates['Comparison_to_Statewide'] = region_avg_rates['Rate'] - statewide_avg

# Output the regions with the highest rates
top_regions = region_avg_rates.head(10)  # Adjust the number to show more or fewer top regions
print("Top regions with the highest asthma ED visit rates:")
print(top_regions)

# Visualize the results
plt.figure(figsize=(10, 6))
plt.barh(top_regions['Geography'], top_regions['Rate'], color='skyblue')
plt.axvline(statewide_avg, color='red', linestyle='--', label=f'Statewide Avg: {statewide_avg:.2f}')
plt.xlabel('Asthma ED Visit Rate')
plt.ylabel('Geography')
plt.title('Top Regions with Highest Asthma ED Visit Rates vs Statewide Average')
plt.legend()
plt.gca().invert_yaxis()
plt.show()

import pandas as pd
import matplotlib.pyplot as plt

# Sample data
data = pd.DataFrame({
    'Age': [5, 12, 25, 40, 65, 80],
    'Strata': ['Urban', 'Rural', 'Urban', 'Rural', 'Urban', 'Rural']
})

# Define age groups
bins = [0, 18, 35, 50, 65, 100]
labels = ['0-18', '19-35', '36-50', '51-65', '66+']
data['Age_Group'] = pd.cut(data['Age'], bins=bins, labels=labels, right=False)

# Encode strata (e.g., Urban/Rural) into numerical values
data['Strata_Code'] = pd.factorize(data['Strata'])[0]

print(data)

# Plotting
fig, ax = plt.subplots(1, 2, figsize=(12, 6))

# Age group distribution plot
data['Age_Group'].value_counts().plot(kind='bar', ax=ax[0], color='skyblue')
ax[0].set_title('Age Group Distribution')
ax[0].set_xlabel('Age Group')
ax[0].set_ylabel('Count')

# Strata distribution plot
data['Strata'].value_counts().plot(kind='bar', ax=ax[1], color='salmon')
ax[1].set_title('Strata Distribution')
ax[1].set_xlabel('Strata')
ax[1].set_ylabel('Count')

plt.tight_layout()
plt.show()
